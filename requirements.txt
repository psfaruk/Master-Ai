# Master-Ai — Python edition
# FastAPI + httpx + asyncio, mirror of the original Next.js/TypeScript dashboard.
#
# All versions are pinned for reproducible Railway builds. Every package here
# ships pre-built Linux x86_64 wheels on PyPI — no C compilation needed, so
# the Dockerfile does NOT install build-essential (which OOMs Railway's
# smallest plan).

# Web framework — FastAPI + ASGI server.
# Plain `uvicorn` (not `uvicorn[standard]`) — the `standard` extras pull in
# `httptools` and `uvloop` which sometimes need to compile against glibc.
# Plain uvicorn falls back to h11+asyncio which is plenty fast for this
# workload (the bottleneck is the 3 upstream HTTP fetches, not the parser).
fastapi==0.115.6
uvicorn==0.34.0

# Template engine for the dashboard UI.
jinja2==3.1.5
itsdangerous==2.2.0

# Async HTTP client — replaces Node's `fetch()`. Used by every module that
# talks to the 3 upstream Quotex signal apps.
httpx==0.28.1

# TLS root certificates — httpx uses this for TLS verification when the
# base Docker image (python:3.12-slim) doesn't ship ca-certificates. We
# install it explicitly so we never have to `apt-get install ca-certificates`
# (which is what was timing out the Railway build).
certifi==2024.12.14

# Pydantic v2 — bundled with FastAPI 0.100+ but we pin it for stability.
pydantic==2.10.4

# Misc utilities.
python-dateutil==2.9.0.post0
